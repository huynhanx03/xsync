# Product-Grade DLHT Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a production-grade `DLHT[K,V]` concurrent hash table to `xsync` with one root public file and a maintainable `internal/dlht` core.

**Architecture:** The root package exposes `DLHT` in `dlht.go` only. All state machines, DWCAS, bucket/index layout, resize, iteration, and tests for internals live under `internal/dlht`. The implementation follows the DLHT paper's bounded cache-line chaining and per-bin cooperative resize, while mirroring `xsync.Map` public method names.

**Tech Stack:** Go 1.24+, generics, `hash/maphash`, `sync/atomic`, `unsafe`, Go assembly for amd64/arm64 16-byte CAS, `testing` benchmarks and concurrency stress tests.

package xsync

import (
	"iter"

	"github.com/puzpuzpuz/xsync/v4/internal/dlht"
)

// DLHT is a high-performance concurrent hash table based on the DLHT design.
// It is safe for concurrent use by multiple goroutines.
//
// DLHT follows the simple Map-style API: Load, Store, LoadOrStore,
// LoadAndStore, LoadAndDelete, Delete, Range, All, Size, and Stats.
// It does not expose Compute because arbitrary callbacks do not fit the
// lock-free slot protocol without a separate synchronization design.
type DLHT[K comparable, V any] struct {
	m *dlht.Map[K, V]
}

// DLHTConfig defines configurable DLHT options.
type DLHTConfig struct {
	sizeHint int
}

// DLHTStats contains an approximate snapshot of a DLHT instance.
type DLHTStats struct {
	Bins         uint64
	Links        uint64
	LinkCapacity uint64
	Size         uint64
	Capacity     uint64
	LoadFactor   float64
	Resizing     bool
}

// WithDLHTPresize configures a new DLHT with capacity for sizeHint entries.
// If sizeHint is zero or negative, the value is ignored.
func WithDLHTPresize(sizeHint int) func(*DLHTConfig) {
	return func(c *DLHTConfig) {
		if sizeHint > 0 {
			c.sizeHint = sizeHint
		}
	}
}

// NewDLHT creates a new DLHT instance configured with the given options.
func NewDLHT[K comparable, V any](options ...func(*DLHTConfig)) *DLHT[K, V] {
	c := &DLHTConfig{sizeHint: 16}
	for _, option := range options {
		option(c)
	}
	return &DLHT[K, V]{
		m: dlht.New[K, V](dlht.Options{InitialSize: uint64(c.sizeHint)}),
	}
}

func (m *DLHT[K, V]) Load(key K) (value V, ok bool) {
	return m.m.Load(key)
}

func (m *DLHT[K, V]) Store(key K, value V) {
	m.m.Store(key, value)
}

func (m *DLHT[K, V]) LoadOrStore(key K, value V) (actual V, loaded bool) {
	return m.m.LoadOrStore(key, value)
}

func (m *DLHT[K, V]) LoadAndStore(key K, value V) (actual V, loaded bool) {
	return m.m.LoadAndStore(key, value)
}

func (m *DLHT[K, V]) LoadAndDelete(key K) (value V, loaded bool) {
	return m.m.LoadAndDelete(key)
}

func (m *DLHT[K, V]) Delete(key K) {
	m.m.Delete(key)
}

func (m *DLHT[K, V]) Range(f func(key K, value V) bool) {
	m.m.Range(f)
}

func (m *DLHT[K, V]) All() iter.Seq2[K, V] {
	return m.m.All()
}

func (m *DLHT[K, V]) Size() int {
	return int(m.m.Size())
}

func (m *DLHT[K, V]) Stats() DLHTStats {
	s := m.m.Stats()
	return DLHTStats{
		Bins:         s.Bins,
		Links:        s.Links,
		LinkCapacity: s.LinkCapacity,
		Size:         s.Size,
		Capacity:     s.Capacity,
		LoadFactor:   s.LoadFactor,
		Resizing:     s.Resizing,
	}
}

const (

import "hash/maphash"

// New creates a DLHT map.
func New[K comparable, V any](opts Options) *Map[K, V] {
	if opts.InitialSize == 0 {
		opts.InitialSize = defaultBinCount
	}
	seed := maphash.MakeSeed()
	m := &Map[K, V]{
		hashSeed: seed,
		intKey:   detectIntKey[K](),
	}
	m.intSeed = deriveIntSeed(seed)
	m.active.Store(newIndex[K, V](nextPowerOf2(opts.InitialSize)))
	return m
}

package dlht

import (
	"hash/maphash"
	"math/bits"
	"unsafe"
)

func nextPowerOf2(v uint64) uint64 {
	if v <= 1 {
		return 1
	}
	return 1 << bits.Len64(v-1)
}

func detectIntKey[K comparable]() bool {
	var zero K
	switch any(zero).(type) {
	case int, uint, uintptr, int64, uint64, int32, uint32, int16, uint16, int8, uint8:
		return true
	default:
		return false
	}
}

func deriveIntSeed(seed maphash.Seed) uint64 {
	var h maphash.Hash
	h.SetSeed(seed)
	h.WriteByte(0)
	return h.Sum64()
}

func hashUint64(seed, v uint64) uint64 {
	hi, lo := bits.Mul64(v^seed, 0x2d358dccaa6c78a5)
	hi2, lo2 := bits.Mul64(hi^lo, 0x8bb84b93962eacc9)
	return hi2 ^ lo2
}

func toUint64[K comparable](k K) uint64 {
	switch unsafe.Sizeof(k) {
	case 8:
		return *(*uint64)(unsafe.Pointer(&k))
	case 4:
		return uint64(*(*uint32)(unsafe.Pointer(&k)))
	case 2:
		return uint64(*(*uint16)(unsafe.Pointer(&k)))
	case 1:
		return uint64(*(*uint8)(unsafe.Pointer(&k)))
	default:
		panic("unreachable")
	}
}

func (m *Map[K, V]) hashKey(key K) uint64 {
	if m.intKey {
		return hashUint64(m.intSeed, toUint64(key))
	}
	return maphash.Comparable(m.hashSeed, key)
}

const (
	evenTransferSentinel = 0xfffffffffffffffe
	oddTransferSentinel  = 0xffffffffffffffff
)

func transferSentinel(binIndex uint64) uint64 {
	if binIndex&1 == 0 {
		return oddTransferSentinel
	}
	return evenTransferSentinel
}
```

package dlht

import (
	"sync/atomic"
	"unsafe"
)

const (
	binNoTransfer   = uint8(0)
	binInTransfer   = uint8(1)
	binDoneTransfer = uint8(2)
)

const (
	slotInvalid = uint8(0)
	slotTrying  = uint8(1)
	slotValid   = uint8(2)
)

const (
	versionShift   = 32
	binStateShift  = 30
	binStateMask   = uint64(0x3) << binStateShift
	slotStatesMask = uint64(1<<30) - 1
)

func (h header) getBinState() uint8 {
	return uint8((uint64(h) & binStateMask) >> binStateShift)
}

func (h header) getSlotStates() uint32 {
	return uint32(uint64(h) & slotStatesMask)
}

func (h header) getSlotState(slotIdx int) uint8 {
	shift := uint(slotIdx*2) & 63
	return uint8((h.getSlotStates() >> shift) & 0x3)
}

func (h header) setBinState(state uint8) header {
	raw := uint64(h) &^ binStateMask
	raw |= uint64(state&0x3) << binStateShift
	return header(raw)
}

func (h header) setSlotStateAndVersion(slotIdx int, state uint8) header {
	shift := uint(slotIdx*2) & 63
	mask := uint64(0x3) << shift
	val := uint64(state&0x3) << shift
	raw := (uint64(h) &^ mask) | val
	raw += 1 << versionShift
	return header(raw)
}

func atomicLoadHeader(addr *header) header {
	return header(atomic.LoadUint64((*uint64)(unsafe.Pointer(addr))))
}

func atomicStoreHeader(addr *header, value header) {
	atomic.StoreUint64((*uint64)(unsafe.Pointer(addr)), uint64(value))
}

func atomicCASHeader(addr *header, old, next header) bool {
	return atomic.CompareAndSwapUint64((*uint64)(unsafe.Pointer(addr)), uint64(old), uint64(next))
}

func atomicOrHeader(addr *header, value uint64) header {
	return header(atomic.OrUint64((*uint64)(unsafe.Pointer(addr)), value))
}

func atomicAndHeader(addr *header, mask uint64) {
	atomic.AndUint64((*uint64)(unsafe.Pointer(addr)), mask)
}

func atomicLoadSlotPtr[K comparable, V any](s *slot[K, V]) *entry[K, V] {
	return (*entry[K, V])(atomic.LoadPointer((*unsafe.Pointer)(unsafe.Pointer(&s.ptr))))
}

func atomicStoreSlotPtr[K comparable, V any](s *slot[K, V], e *entry[K, V]) {
	atomic.StorePointer((*unsafe.Pointer)(unsafe.Pointer(&s.ptr)), unsafe.Pointer(e))
}

func (h header) validMask3() uint32 {
	return (uint32(h) >> 1) & 0x15
}

func (h header) validMask4(base uint) uint32 {
	return (uint32(h) >> ((base << 1) + 1)) & 0x55
}

func (h header) invalidMask3() uint32 {
	s := uint32(h) & 0x3f
	return ^(s | (s >> 1)) & 0x15
}

func (h header) invalidMask4(base uint) uint32 {
	s := (uint32(h) >> (base << 1)) & 0xff
	return ^(s | (s >> 1)) & 0x55
}

func boolBit(b bool) uint32 {
	if b {
		return 1
	}
	return 0
}

package dlht

import (
	"math/bits"
	"unsafe"
)

func newIndex[K comparable, V any](numBins uint64) *index[K, V] {
	numBins = nextPowerOf2(numBins)
	if numBins < defaultBinCount {
		numBins = defaultBinCount
	}
	linkCount := numBins / 8
	if linkCount < minLinkArenaSize {
		linkCount = minLinkArenaSize
	}
	idx := &index[K, V]{
		mask:         numBins - 1,
		bins:         make([]primaryBucket[K, V], numBins),
		links:        make([]linkBucket[K, V], linkCount),
		linkCapacity: uint32(linkCount),
	}
	return idx
}

func (idx *index[K, V]) getNextIndex() *index[K, V] {
	return idx.indexNext.Load()
}

func (idx *index[K, V]) getBin(hash uint64) *primaryBucket[K, V] {
	return idx.getBinByIndex(hash & idx.mask)
}

func (idx *index[K, V]) getBinByIndex(binIndex uint64) *primaryBucket[K, V] {
	base := unsafe.SliceData(idx.bins)
	return (*primaryBucket[K, V])(unsafe.Add(unsafe.Pointer(base), uintptr(binIndex)*unsafe.Sizeof(*base)))
}

func (idx *index[K, V]) getLinkBucket(linkIndex uint32) *linkBucket[K, V] {
	base := unsafe.SliceData(idx.links)
	return (*linkBucket[K, V])(unsafe.Add(unsafe.Pointer(base), uintptr(linkIndex-1)*unsafe.Sizeof(*base)))
}

func (pb *primaryBucket[K, V]) slotAt(i int) *slot[K, V] {
	return (*slot[K, V])(unsafe.Add(unsafe.Pointer(&pb.slots), uintptr(i)*unsafe.Sizeof(pb.slots[0])))
}

func (lb *linkBucket[K, V]) slotAt(i int) *slot[K, V] {
	return (*slot[K, V])(unsafe.Add(unsafe.Pointer(&lb.slots), uintptr(i)*unsafe.Sizeof(lb.slots[0])))
}

func (idx *index[K, V]) getSlotByIndex(pb *primaryBucket[K, V], slotIndex int) *slot[K, V] {
	if slotIndex < primarySlots {
		return pb.slotAt(slotIndex)
	}
	linkIdx := slotIndex - primarySlots
	bucketIdx := linkIdx >> 2
	slotInBucket := linkIdx & 3
	lm := atomicLoadLinkMeta(&pb.linkMeta)
	addr := lm.resolveLink(bucketIdx)
	if addr == noLink {
		return nil
	}
	return idx.getLinkBucket(addr).slotAt(slotInBucket)
}

func chooseInsertSlot[K comparable, V any](idx *index[K, V], pb *primaryBucket[K, V]) int {
	h := atomicLoadHeader(&pb.header)
	if mask := h.invalidMask3(); mask != 0 {

package dlht

import "sync/atomic"

const (
	pairStartShift = 32
	noLink         = uint32(0)
)

func newLinkMeta(pairStart, single uint32) linkMeta {
	return linkMeta((uint64(pairStart) << pairStartShift) | uint64(single))
}

func (lm linkMeta) getPairStart() uint32 {
	return uint32(uint64(lm) >> pairStartShift)
}

func (lm linkMeta) getSingle() uint32 {
	return uint32(uint64(lm))
}

func (lm linkMeta) setPairStart(pairStart uint32) linkMeta {
	return newLinkMeta(pairStart, lm.getSingle())
}

func (lm linkMeta) setSingle(single uint32) linkMeta {
	return newLinkMeta(lm.getPairStart(), single)
}

func (lm linkMeta) resolveLink(linkIdx int) uint32 {
	switch linkIdx {
	case 0:
		return lm.getSingle()
	case 1:
		return lm.getPairStart()
	case 2:
		pair := lm.getPairStart()
		if pair == noLink {
			return noLink
		}
		return pair + 1
	default:
		return noLink
	}
}

func atomicLoadLinkMeta(addr *linkMeta) linkMeta {
	return linkMeta(atomic.LoadUint64((*uint64)(addr)))
}

func atomicCASLinkMeta(addr *linkMeta, old, next linkMeta) bool {
	return atomic.CompareAndSwapUint64((*uint64)(addr), uint64(old), uint64(next))
}

func (idx *index[K, V]) allocateLink() uint32 {
	next := atomic.AddUint32(&idx.linkNext, 1)
	if next > atomic.LoadUint32(&idx.linkCapacity) {
		return noLink
	}
	return next
}

func (idx *index[K, V]) allocateLinkPair() uint32 {
	next := atomic.AddUint32(&idx.linkNext, 2) - 1
	if next+1 > atomic.LoadUint32(&idx.linkCapacity) {
		return noLink
	}
	return next
}

func (idx *index[K, V]) attachSingle(pb *primaryBucket[K, V]) uint32 {
	for {
		old := atomicLoadLinkMeta(&pb.linkMeta)
		if single := old.getSingle(); single != noLink {
			return single
		}
		link := idx.allocateLink()
		if link == noLink {
			return noLink
		}
		if atomicCASLinkMeta(&pb.linkMeta, old, old.setSingle(link)) {
			return link
		}
	}
}

func (idx *index[K, V]) attachPair(pb *primaryBucket[K, V]) uint32 {
	for {
		old := atomicLoadLinkMeta(&pb.linkMeta)
		if pair := old.getPairStart(); pair != noLink {
			return pair
		}
		pair := idx.allocateLinkPair()
		if pair == noLink {
			return noLink
		}
		if atomicCASLinkMeta(&pb.linkMeta, old, old.setPairStart(pair)) {
			return pair
		}
	}
}

	}

		goto retry
	}

	return zero, false

func findEntryInBin[K comparable, V any](idx *index[K, V], pb *primaryBucket[K, V], h header, key K, hash uint64) *entry[K, V] {
	bitmap := matchPrimaryHashes(pb, h, hash)
	for bitmap != 0 {
		slotIdx := bits.TrailingZeros32(bitmap) >> 1
		candidate := atomicLoadSlotPtr(pb.slotAt(slotIdx))
		if candidate != nil && candidate.key == key {
			return candidate
		}
		bitmap &= bitmap - 1
	}
	return scanLinksForEntry(idx, pb, h, key, hash)
}

func matchPrimaryHashes[K comparable, V any](pb *primaryBucket[K, V], h header, hash uint64) uint32 {
	bitmap := (boolBit(pb.slots[0].hash == hash) << 0) |
		(boolBit(pb.slots[1].hash == hash) << 2) |
		(boolBit(pb.slots[2].hash == hash) << 4)
	return bitmap & (uint32(h) >> 1)
}

func scanLinksForEntry[K comparable, V any](idx *index[K, V], pb *primaryBucket[K, V], h header, key K, hash uint64) *entry[K, V] {

			return e
		}

			return e

			return e
		}

}

	return m.store(key, value, false, false)
}

func (m *Map[K, V]) store(key K, value V, onlyIfAbsent bool, discardOld bool) (actual V, loaded bool) {
	hash := m.hashKey(key)
	idx := m.active.Load()
retry:
	pb := idx.getBin(hash)
	h0 := atomicLoadHeader(&pb.header)
	switch h0.getBinState() {
	case binDoneTransfer:
		idx = idx.getNextIndex()
		goto retry
	case binInTransfer:
		runtime.Gosched()
		idx = m.active.Load()
		goto retry
	}
	if slotIdx, s, e := findSlotInBin(idx, pb, h0, key, hash); e != nil {
		if atomicLoadHeader(&pb.header) != h0 {
			goto retry
		}
		if onlyIfAbsent {
			return e.value, true
		}
		next := &entry[K, V]{key: key, value: value}
		if dwcasSlot(s, hash, e, hash, next) {
			if discardOld {
				return value, true
			}
			return e.value, true
		}
		_ = slotIdx
		goto retry
	}
	if atomicLoadHeader(&pb.header) != h0 {
		goto retry
	}
	slotIndex := chooseInsertSlot(idx, pb)
	if slotIndex < 0 {
		m.triggerResize()
		m.helpResize()
		idx = m.active.Load()
		goto retry
	}
	reserved := h0.setSlotStateAndVersion(slotIndex, slotTrying)
	if !atomicCASHeader(&pb.header, h0, reserved) {
		goto retry
	}
	s := idx.getSlotByIndex(pb, slotIndex)
	if s == nil {
		m.unreserveSlot(idx, pb, slotIndex)
		goto retry
	}
	s.hash = hash
	atomicStoreSlotPtr(s, &entry[K, V]{key: key, value: value})
	if atomicCASHeader(&pb.header, reserved, reserved.setSlotStateAndVersion(slotIndex, slotValid)) {
		return value, false
	}
	m.unreserveSlot(idx, pb, slotIndex)
	goto retry
}

func findSlotInBin[K comparable, V any](idx *index[K, V], pb *primaryBucket[K, V], h header, key K, hash uint64) (int, *slot[K, V], *entry[K, V]) {
	for i := uint32(matchPrimaryHashes(pb, h, hash)); i != 0; i &= i - 1 {
		slotIdx := trailingSlot(i)
		s := pb.slotAt(slotIdx)
		candidate := atomicLoadSlotPtr(s)
		if candidate != nil && candidate.key == key {
			return slotIdx, s, candidate
		}
	}
	return findSlotInLinks(idx, pb, h, key, hash)
}

func findSlotInLinks[K comparable, V any](idx *index[K, V], pb *primaryBucket[K, V], h header, key K, hash uint64) (int, *slot[K, V], *entry[K, V]) {
	lm := atomicLoadLinkMeta(&pb.linkMeta)
	if single := lm.getSingle(); single != noLink {
		if i, s, e := findSlotInLinkBucket(idx.getLinkBucket(single), h.validMask4(primarySlots), key, hash, primarySlots); e != nil {
			return i, s, e
		}
	}
	if pair := lm.getPairStart(); pair != noLink {
		if i, s, e := findSlotInLinkBucket(idx.getLinkBucket(pair), h.validMask4(primarySlots+linkSlots), key, hash, primarySlots+linkSlots); e != nil {
			return i, s, e
		}
		if i, s, e := findSlotInLinkBucket(idx.getLinkBucket(pair+1), h.validMask4(primarySlots+2*linkSlots), key, hash, primarySlots+2*linkSlots); e != nil {
			return i, s, e
		}
	}
	return -1, nil, nil
}

func findSlotInLinkBucket[K comparable, V any](lb *linkBucket[K, V], mask uint32, key K, hash uint64, base int) (int, *slot[K, V], *entry[K, V]) {
	eq := (boolBit(lb.slots[0].hash == hash) << 0) |
		(boolBit(lb.slots[1].hash == hash) << 2) |
		(boolBit(lb.slots[2].hash == hash) << 4) |
		(boolBit(lb.slots[3].hash == hash) << 6)
	eq &= mask
	for eq != 0 {
		slotIdx := trailingSlot(eq)
		s := lb.slotAt(slotIdx)
		candidate := atomicLoadSlotPtr(s)
		if candidate != nil && candidate.key == key {
			return base + slotIdx, s, candidate
		}
		eq &= eq - 1
	}
	return -1, nil, nil
}

func trailingSlot(mask uint32) int {
	return bitsTrailingZeros32(mask) >> 1
}

func (m *Map[K, V]) unreserveSlot(idx *index[K, V], pb *primaryBucket[K, V], slotIndex int) bool {
	for {
		h := atomicLoadHeader(&pb.header)
		if h.getBinState() != binNoTransfer || h.getSlotState(slotIndex) != slotTrying {
			return false
		}
		if s := idx.getSlotByIndex(pb, slotIndex); s != nil {
			atomicStoreSlotPtr(s, nil)
		}
		next := h.setSlotStateAndVersion(slotIndex, slotInvalid)
		if atomicCASHeader(&pb.header, h, next) {
			return true
		}
	}
}

import (
	"math/bits"
	"sync/atomic"
)

func (m *Map[K, V]) Size() uint64 {
	return m.active.Load().validSlotCount()
}

func (m *Map[K, V]) Stats() Stats {
	idx := m.active.Load()
	bins := uint64(len(idx.bins))
	linkCapacity := uint64(len(idx.links))
	links := uint64(atomic.LoadUint32(&idx.linkNext))
	size := idx.validSlotCount()
	capacity := bins * maxSlotsPerBin
	var loadFactor float64
	if capacity != 0 {
		loadFactor = float64(size) / float64(capacity)
	}
	return Stats{
		Bins:         bins,
		Links:        links,
		LinkCapacity: linkCapacity,
		Size:         size,
		Capacity:     capacity,
		LoadFactor:   loadFactor,
		Resizing:     m.resizeCtx.Load() != nil,
	}
}

func (idx *index[K, V]) validSlotCount() uint64 {
	const validMask uint32 = 0x2aaaaaaa
	var n uint64
	for i := range idx.bins {
		h := atomicLoadHeader(&idx.bins[i].header)
		n += uint64(bits.OnesCount32(uint32(h) & validMask))
	}
	return n
}

package dlht

import "iter"

func (m *Map[K, V]) Range(yield func(K, V) bool) {
	idx := m.active.Load()
	m.rangeFrom(idx, yield)
}

func (m *Map[K, V]) All() iter.Seq2[K, V] {
	return func(yield func(K, V) bool) {
		m.Range(yield)
	}
}

func (m *Map[K, V]) rangeFrom(idx *index[K, V], yield func(K, V) bool) bool {
	for binIdx := range idx.bins {
		if !m.scanBin(idx, uint64(binIdx), yield) {
			return false
		}
	}
	return true
}

func (m *Map[K, V]) scanBin(idx *index[K, V], binIdx uint64, yield func(K, V) bool) bool {
retry:
	pb := idx.getBinByIndex(binIdx)
	h0 := atomicLoadHeader(&pb.header)
	switch h0.getBinState() {
	case binDoneTransfer:
		next := idx.getNextIndex()
		if next == nil {
			return true
		}
		return m.scanShard(next, binIdx, idx.mask+1, yield)
	case binInTransfer:
		return true
	}
	if !readPrimarySlots(pb, h0.validMask3(), yield) {
		return false
	}
	lm := atomicLoadLinkMeta(&pb.linkMeta)
	if single := lm.getSingle(); single != noLink {
		if !readLinkSlots(idx.getLinkBucket(single), h0.validMask4(primarySlots), yield) {
			return false
		}
	}
	if pair := lm.getPairStart(); pair != noLink {
		if !readLinkSlots(idx.getLinkBucket(pair), h0.validMask4(primarySlots+linkSlots), yield) {
			return false
		}
		if !readLinkSlots(idx.getLinkBucket(pair+1), h0.validMask4(primarySlots+2*linkSlots), yield) {
			return false
		}
	}
	if atomicLoadHeader(&pb.header) != h0 {
		goto retry
	}
	return true
}

func (m *Map[K, V]) scanShard(idx *index[K, V], oldBinIdx uint64, oldLen uint64, yield func(K, V) bool) bool {
	for binIdx := oldBinIdx; binIdx < uint64(len(idx.bins)); binIdx += oldLen {
		if !m.scanBin(idx, binIdx, yield) {
			return false
		}
	}
	return true
}

func readPrimarySlots[K comparable, V any](pb *primaryBucket[K, V], mask uint32, yield func(K, V) bool) bool {
	for mask != 0 {
		slotIdx := trailingSlot(mask)
		if e := atomicLoadSlotPtr(pb.slotAt(slotIdx)); e != nil {
			if !yield(e.key, e.value) {
				return false
			}
		}
		mask &= mask - 1
	}
	return true
}

func readLinkSlots[K comparable, V any](lb *linkBucket[K, V], mask uint32, yield func(K, V) bool) bool {
	for mask != 0 {
		slotIdx := trailingSlot(mask)
		if e := atomicLoadSlotPtr(lb.slotAt(slotIdx)); e != nil {
			if !yield(e.key, e.value) {
				return false
			}
		}
		mask &= mask - 1
	}
	return true
}

package dlht

import (
	"math/bits"
	"runtime"
	"sync/atomic"
	"unsafe"
)

const (
	resizeChunkSize = 1024
	resizeInactive  = uint32(0)
	resizeReserved  = uint32(1)
	resizeActive    = uint32(2)
)

func (m *Map[K, V]) triggerResize() bool {
	reservedCtx := &resizeContext[K, V]{active: resizeReserved}
	if !m.resizeCtx.CompareAndSwap(nil, reservedCtx) {
		return false
	}
	oldIdx := m.active.Load()
	oldSize := uint64(len(oldIdx.bins))
	growth := uint64(2)
	if oldSize < 4096 {
		growth = 8
	} else if oldSize < 4*1024*1024 {
		growth = 4
	}
	newIdx := newIndex[K, V](oldSize * growth)
	reserveLinks := uint32(len(oldIdx.links))*4/3 + 1
	if reserveLinks < uint32(len(newIdx.links)) {
		newIdx.linkCapacity = uint32(len(newIdx.links)) - reserveLinks
	}
	oldIdx.indexNext.Store(newIdx)
	ctx := &resizeContext[K, V]{
		oldIndex:       oldIdx,
		newIndex:       newIdx,
		numBins:        oldSize,
		numChunks:      (oldSize + resizeChunkSize - 1) / resizeChunkSize,
		active:         resizeActive,
		linkNextResize: uint32(len(newIdx.links)),
	}
	m.resizeCtx.Store(ctx)
	m.helpWithChunks(ctx)
	return true
}

func (m *Map[K, V]) helpResize() {
	for {
		ctx := m.resizeCtx.Load()
		if ctx == nil {
			return
		}
		switch atomic.LoadUint32(&ctx.active) {
		case resizeInactive:
			return
		case resizeReserved:
			runtime.Gosched()
		case resizeActive:
			m.helpWithChunks(ctx)
		}
	}
}

func (m *Map[K, V]) helpWithChunks(ctx *resizeContext[K, V]) {
	for {
		chunk := atomic.AddUint64(&ctx.chunkClaimed, 1) - 1
		if chunk >= ctx.numChunks {
			return
		}
		m.transferChunk(ctx, chunk)
		done := atomic.AddUint64(&ctx.chunksDone, 1)
		if done == ctx.numChunks {
			atomic.StoreUint32(&ctx.newIndex.linkCapacity, ctx.linkNextResize-1)
			m.active.Store(ctx.newIndex)
			atomic.StoreUint32(&ctx.active, resizeInactive)
			ctx.oldIndex = nil
			ctx.newIndex = nil
			m.resizeCtx.Store(nil)
			return
		}
	}
}

func (m *Map[K, V]) transferChunk(ctx *resizeContext[K, V], chunk uint64) {
	start := chunk * resizeChunkSize
	end := start + resizeChunkSize
	if end > ctx.numBins {
		end = ctx.numBins
	}
	for i := start; i < end; i++ {
		m.transferBin(ctx, i)
	}
}

func (m *Map[K, V]) transferBin(ctx *resizeContext[K, V], binIndex uint64) {
	oldBin := ctx.oldIndex.getBinByIndex(binIndex)
	h := atomicOrHeader(&oldBin.header, uint64(binInTransfer)<<binStateShift)
	sentinel := transferSentinel(binIndex)
	m.transferValidSlots(ctx, h.validMask3(), oldBin.slotAt(0), sentinel)
	lm := oldBin.linkMeta
	if single := lm.getSingle(); single != noLink {
		m.transferValidSlots(ctx, h.validMask4(primarySlots), ctx.oldIndex.getLinkBucket(single).slotAt(0), sentinel)
	}
	if pair := lm.getPairStart(); pair != noLink {
		m.transferValidSlots(ctx, h.validMask4(primarySlots+linkSlots), ctx.oldIndex.getLinkBucket(pair).slotAt(0), sentinel)
		m.transferValidSlots(ctx, h.validMask4(primarySlots+2*linkSlots), ctx.oldIndex.getLinkBucket(pair+1).slotAt(0), sentinel)
	}
	atomicStoreHeader(&oldBin.header, h.setBinState(binDoneTransfer))
}

func (m *Map[K, V]) transferValidSlots(ctx *resizeContext[K, V], mask uint32, base *slot[K, V], sentinel uint64) {
	for mask != 0 {
		slotIdx := bits.TrailingZeros32(mask) >> 1
		s := (*slot[K, V])(unsafe.Add(unsafe.Pointer(base), uintptr(slotIdx)*unsafe.Sizeof(*base)))
		hash := s.hash
		atomic.StoreUint64(&s.hash, sentinel)
		if e := atomicLoadSlotPtr(s); e != nil {
			ctx.insertIntoNewIndex(ctx.newIndex, hash, e)
		}
		mask &= mask - 1
	}
}

func (ctx *resizeContext[K, V]) insertIntoNewIndex(idx *index[K, V], hash uint64, e *entry[K, V]) {
	pb := idx.getBin(hash)
	slotIndex := ctx.findFreeSlotOrChain(pb)
	if slotIndex < 0 {
		panic("xsync DLHT resize destination unexpectedly full")
	}
	s := idx.getSlotByIndex(pb, slotIndex)
	s.hash = hash
	s.ptr = e
	pb.header = pb.header.setSlotStateAndVersion(slotIndex, slotValid)
}

func (ctx *resizeContext[K, V]) findFreeSlotOrChain(pb *primaryBucket[K, V]) int {
	h := pb.header
	if mask := h.invalidMask3(); mask != 0 {
		return bits.TrailingZeros32(mask) >> 1
	}
	lm := pb.linkMeta
	if lm.getSingle() == noLink {
		if ctx.attachSingleForResize(pb) != noLink {
			return primarySlots
		}
	} else if mask := h.invalidMask4(primarySlots); mask != 0 {
		return primarySlots + (bits.TrailingZeros32(mask) >> 1)
	}
	if lm.getPairStart() == noLink {
		if ctx.attachPairForResize(pb) != noLink {
			return primarySlots + linkSlots
		}
	} else {
		if mask := h.invalidMask4(primarySlots + linkSlots); mask != 0 {
			return primarySlots + linkSlots + (bits.TrailingZeros32(mask) >> 1)
		}
		if mask := h.invalidMask4(primarySlots + 2*linkSlots); mask != 0 {
			return primarySlots + 2*linkSlots + (bits.TrailingZeros32(mask) >> 1)
		}
	}
	return -1
}

func (ctx *resizeContext[K, V]) nextSingleLinkForResize() uint32 {
	next := atomic.AddUint32(&ctx.linkNextResize, ^uint32(0))
	if next == 0 {
		return noLink
	}
	return next
}

func (ctx *resizeContext[K, V]) nextLinkPairForResize() uint32 {
	next := atomic.AddUint32(&ctx.linkNextResize, ^uint32(1))
	if next <= 1 {
		return noLink
	}
	return next
}

func (ctx *resizeContext[K, V]) attachSingleForResize(pb *primaryBucket[K, V]) uint32 {
	link := ctx.nextSingleLinkForResize()
	pb.linkMeta = pb.linkMeta.setSingle(link)
	return link
}

func (ctx *resizeContext[K, V]) attachPairForResize(pb *primaryBucket[K, V]) uint32 {
	pair := ctx.nextLinkPairForResize()
	pb.linkMeta = pb.linkMeta.setPairStart(pair)
	return pair
}

package dlht

import (
	"sync"
	"sync/atomic"
	"testing"
)

type timedOp struct {
	kind  byte
	key   int
	value int
	start uint64
	end   uint64
	ok    bool
	old   int
}

func TestPerKeyLinearizabilitySmallHistory(t *testing.T) {
	m := New[int, int](Options{InitialSize: 2})
	var clock atomic.Uint64
	var wg sync.WaitGroup
	hist := make([][]timedOp, 4)
	for worker := 0; worker < 4; worker++ {
		wg.Add(1)
		go func(worker int) {
			defer wg.Done()
			local := make([]timedOp, 0, 64)
			for i := 0; i < 64; i++ {
				key := i % 2
				value := worker*1000 + i
				op := timedOp{key: key, value: value, start: clock.Add(1)}
				switch i % 4 {
				case 0:
					m.Store(key, value)
					op.kind = 's'
					op.ok = true
				case 1:
					op.kind = 'l'
					op.old, op.ok = m.Load(key)
				case 2:
					op.kind = 'u'
					op.old, op.ok = m.LoadAndStore(key, value)
				case 3:
					op.kind = 'd'
					op.old, op.ok = m.LoadAndDelete(key)
				}
				op.end = clock.Add(1)
				local = append(local, op)
			}
			hist[worker] = local
		}(worker)
	}
	wg.Wait()
	for key := 0; key < 2; key++ {
		lastEnd := uint64(0)
		present := false
		value := 0
		for {
			var chosen *timedOp
			for wi := range hist {
				for oi := range hist[wi] {
					op := &hist[wi][oi]
					if op.key == key && op.start > lastEnd && (chosen == nil || op.end < chosen.end) {
						chosen = op
					}
				}
			}
			if chosen == nil {
				break
			}
			if chosen.start > lastEnd {
				switch chosen.kind {
				case 's':
					present = true
					value = chosen.value
				case 'l':
					if chosen.ok && (!present || chosen.old != value) {
						t.Fatalf("key %d load old=%d present=%v value=%d", key, chosen.old, present, value)
					}
				case 'u':
					if chosen.ok != present {
						t.Fatalf("key %d update ok=%v present=%v", key, chosen.ok, present)
					}
					if chosen.ok && chosen.old != value {
						t.Fatalf("key %d update old=%d value=%d", key, chosen.old, value)
					}
					present = true
					value = chosen.value
				case 'd':
					if chosen.ok != present {
						t.Fatalf("key %d delete ok=%v present=%v", key, chosen.ok, present)
					}
					if chosen.ok && chosen.old != value {
						t.Fatalf("key %d delete old=%d value=%d", key, chosen.old, value)
					}
					present = false
				}
			}
			lastEnd = chosen.end
		}
	}
}

### DLHT

A `DLHT` is a high-performance concurrent hash table based on the DLHT design. It exposes the same simple map-style operations as `Map`, while using bounded cache-line bucket chains, lock-free reads, immediate delete slot reuse, double-word CAS updates, and cooperative per-bin resizing internally.

```go
m := xsync.NewDLHT[string, string](xsync.WithDLHTPresize(1024))
m.Store("foo", "bar")
v, ok := m.Load("foo")
_, loaded := m.LoadAndDelete("foo")
_, _ = v, ok
_ = loaded
```

`DLHT` is a separate type and does not replace `Map`. Use it when write scalability, delete-heavy churn, and bounded overflow behavior matter more than the broader `Map` API surface. `DLHT` intentionally does not expose `Compute`.
